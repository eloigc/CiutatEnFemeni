const map = L.map('map', { zoomControl: false, tap: false }).setView([41.390, 2.170], 13);
L.tileLayer('https://{s}.basemaps.cartocdn.com/light_all/{z}/{x}/{y}{r}.png', { attribution: '© CARTO' }).addTo(map);
const carousel = document.getElementById('carousel');
let allMarkers = [];
function renderArtistes() {
artistes.forEach((artista, aIdx) => {
artista.markers = [];
artista.obres.forEach((obra) => {
const icon = L.divIcon({
className: 'custom-div-icon',
html: `<div class="custom-marker" style="background-image: url('${obra.img}');"></div>`,
iconSize: [32, 32], iconAnchor: [16, 16]
});
const m = L.marker(obra.coords, { icon: icon }).addTo(map);
const popupContent = `
<div class="popup-card">
<strong>${obra.id}.- ${artista.nom}</strong><br>
<strong>${obra.adreca}</strong><br>
<!-- <small>${obra.titol}</small> -->
<img src="${obra.img}">
<a href="https://www.google.com/maps/dir/?api=1&destination=${obra.coords[0]},${obra.coords[1]}" target="_blank" class="btn-route">Com arribar</a>
</div>
`;
m.bindPopup(popupContent, {
offset: [0, -10],
autoPan: true,
autoPanPadding: L.point(50, 150)
});
m.on('click', () => {
seleccionarArtista(aIdx, false);
});
artista.markers.push(m);
allMarkers.push(m);
});
// ── Targeta de l'artista al carrussel ──────────────────────────
const serieText = artista.serie ? artista.serie : '&nbsp;';
const ubicUrl = artista.ubicacionsUrl || '#';
const numUbic = artista.obres.length;
const item = document.createElement('div');
item.className = 'carousel-item';
item.innerHTML = `
<img src="${artista.foto || artista.obres[0].img}" alt="${artista.nom}">
<div class="carousel-item-info">
<span class="carousel-item-nom">${artista.nom}</span>
<span class="carousel-item-serie">${serieText}</span>
<a class="carousel-item-ubicacions"
href="${ubicUrl}"
target="_blank"
onclick="event.stopPropagation();">
${numUbic} UBICACIONS
</a>
</div>
`;
item.addEventListener('click', (e) => {
e.preventDefault();
e.stopPropagation();
seleccionarArtista(aIdx, true);
});
carousel.appendChild(item);
artista.element = item;
});
}
function seleccionarArtista(index, moveMap) {
const artista = artistes[index];
document.querySelectorAll('.carousel-item').forEach(el => el.classList.remove('active'));
artista.element.classList.add('active');
artista.element.scrollIntoView({
behavior: 'smooth',
inline: 'center',
block: 'nearest'
});
document.querySelectorAll('.custom-marker').forEach(el => el.classList.remove('marker-active'));
artista.markers.forEach(m => {
const elementHtml = m.getElement().querySelector('.custom-marker');
if (elementHtml) elementHtml.classList.add('marker-active');
});
if (moveMap) {
const group = new L.featureGroup(artista.markers);
map.fitBounds(group.getBounds(), { padding: [100, 200], maxZoom: 15 });
}
}
function scrollCarousel(dir) {
carousel.scrollBy({ left: dir * 200, behavior: 'smooth' });
}
// ── Inicialització ────────────────────────────────────────────────────
document.getElementById('loading-msg').style.display = 'none';
renderArtistes();
setTimeout(() => seleccionarArtista(0, true), 500);
